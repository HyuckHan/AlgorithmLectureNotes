\section{Part O. Bellman--Ford Algorithm}
\begin{frame}{Bellman--Ford의 약속}
\begin{itemize}\item negative edge 허용\item reachable negative cycle이 없으면 finite shortest paths 계산\item reachable negative cycle 검출 가능\end{itemize}
\begin{alertblock}{표준 알고리즘}각 pass에 모든 edge를 순회한다. queue-based SPFA와 섞지 않는다.\end{alertblock}
\end{frame}
\begin{frame}{Bellman--Ford Pseudocode}
\scriptsize\begin{algorithmic}[1]
\Procedure{BellmanFord}{$G,s$}\State 모든 \(dist\gets\infty,parent\gets\NIL;\ dist[s]\gets0\)
\For{$i\gets1$ to \(|V|-1\)}\State \(changed\gets false\)
\ForAll{$(u,v,w)\in E$}\If{$dist[u]\ne\infty$ \textbf{ and } $dist[v]>dist[u]+w$}
\State update \(dist[v],parent[v]\); \(changed\gets true\)\EndIf\EndFor
\If{\textbf{not} \(changed\)}\State\textbf{break}\EndIf\EndFor
\ForAll{$(u,v,w)\in E$}\If{same relaxation is possible}\State\Return REACHABLE\_NEGATIVE\_CYCLE\EndIf\EndFor
\State\Return \(dist,parent\)\EndProcedure
\end{algorithmic}
\end{frame}
\begin{frame}{왜 \(|V|-1\) Pass인가?}
\begin{itemize}\item reachable negative cycle이 없으면 shortest path를 simple path로 선택 가능\item simple path는 최대 \(|V|-1\) edges\item repeated full passes가 path-relaxation을 전달\end{itemize}
\begin{alertblock}{주의}in-place edge order에 따라 한 pass에서 여러 edge가 연속 반영될 수 있다. pass \(k\)가 정확히 \(k\)-edge path만 뜻하지 않는다.\end{alertblock}
\end{frame}
\begin{frame}{Bellman--Ford 고정 예제}
\small vertices \(s,a,b,c,d\); edge order:
\[
(c,d,2),(a,c,-2),(b,c,3),(s,a,4),(s,b,5),(b,d,6).
\]
\begin{block}{정답}\(dist=[0,4,5,2,4]\), path to \(d\): \(s\to a\to c\to d\).\end{block}
\end{frame}
\begin{frame}{Bellman--Ford Pass Trace}
\[
\begin{array}{c|rrrrr}&s&a&b&c&d\\\hline
initial&0&\infty&\infty&\infty&\infty\\
pass~1&0&4&5&\infty&11\\
pass~2&0&4&5&2&11\\
pass~3&0&4&5&2&4\\
pass~4&0&4&5&2&4
\end{array}
\]
\gtakeaway{pass 4에서 changed=false이므로 early exit.}
\end{frame}
\begin{frame}{Reachable Negative Cycle}
\[
s\to a:1,\qquad a\to b:-2,\qquad b\to a:-2.
\]
\centering\begin{tikzpicture}
\node[pq]at(0,.8){After \(V-1\) passes: one more relaxation is possible};
\only<1>{\node[callout]at(0,-.3){inspect \(a\to b\)};}
\only<2>{\node[callout,draw=red!70!black]at(0,-.3){reachable negative cycle detected};}
\end{tikzpicture}
\end{frame}
\begin{frame}{Bellman--Ford의 DP 해석}
\small\[
D_0(s)=0,\quad D_0(v)=\infty\ (v\ne s)
\]
\[
D_k(v)=\min\left(D_{k-1}(v),\ \min_{(u,v)\in E}\{D_{k-1}(u)+w(u,v)\}\right).
\]
\begin{alertblock}{원본 교정}\(D_{k-1}(v)\), 즉 edge를 추가하지 않는 선택이 반드시 필요하다. 이 식은 separate table의 synchronous DP다.\end{alertblock}
\end{frame}
\begin{frame}{Synchronous DP vs In-Place}
\begin{columns}[T]\column{.48\textwidth}\begin{block}{DP table}pass \(k\)는 오직 row \(k-1\)을 읽는다.\end{block}
\column{.48\textwidth}\begin{block}{표준 in-place BF}같은 pass의 앞선 update를 뒤 edge가 즉시 사용할 수 있다.\end{block}\end{columns}
\end{frame}
\begin{frame}{Bellman--Ford Complexity}
\[
\Theta(VE)\text{ worst-case time},\qquad \Theta(V)\text{ arrays}.
\]
early exit는 쉬운 input을 줄일 수 있지만 worst case는 유지된다. negative-cycle detection pass는 \(O(E)\).
\end{frame}

\end{frame}
