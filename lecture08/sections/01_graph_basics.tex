\begin{frame}{Directed · Undirected · Weighted}
\begin{columns}[T]\column{.32\textwidth}\begin{block}{Undirected}$\{u,v\}=\{v,u\}$\end{block}
\column{.32\textwidth}\begin{block}{Directed}$(u,v)\ne(v,u)$\end{block}
\column{.32\textwidth}\begin{block}{Weighted}$w(u,v)$는 거리·시간·비용·capacity 등 문맥을 가진다.\end{block}\end{columns}
\begin{alertblock}{같은 weight가 아니다}MST는 전체 연결 비용, shortest path는 한 경로의 누적 비용을 최소화한다.\end{alertblock}
\end{frame}
\begin{frame}{Path · Simple Path · Cycle}
\begin{description}\item[Path]연속 edge를 따라가는 vertex sequence\item[Simple path]vertex를 반복하지 않는 path\item[Cycle]시작과 끝이 같고 내부 vertex가 반복되지 않는 닫힌 path\item[Connected]모든 vertex pair 사이에 path가 있는 undirected graph\end{description}
\begin{block}{Connected component}undirected graph에서 서로 도달 가능한 maximal vertex 집합. directed graph의 strong/weak connectivity는 appendix preview로 둔다.\end{block}
\end{frame}
\begin{frame}{Degree와 방향}
\begin{columns}[T]\column{.48\textwidth}\begin{block}{Undirected}$\deg(v)$: $v$에 incident한 edge 수\\$\sum_v\deg(v)=2|E|$\end{block}
\column{.48\textwidth}\begin{block}{Directed}$\operatorname{indeg}(v)$: 들어오는 edge\\$\operatorname{outdeg}(v)$: 나가는 edge\end{block}\end{columns}
\begin{alertblock}{기본 convention}vertex는 $0,\ldots,n-1$ 또는 A,B,\ldots; parallel edge/self-loop는 기본 예제에서 제외; adjacency order를 명시한다.\end{alertblock}
\end{frame}
