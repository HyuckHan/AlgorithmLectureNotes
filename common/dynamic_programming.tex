\newcommand{\dpstate}[1]{\begin{block}{State 의미}#1\end{block}}
\newcommand{\dpdesign}[5]{%
\begin{tikzpicture}[node distance=1.5mm,every node/.style={align=left}]
\node[tree node,minimum width=.92\linewidth,text width=.86\linewidth] (s)
  {\textbf{State}\quad #1};
\node[tree node,minimum width=.92\linewidth,text width=.86\linewidth,below=of s] (t)
  {\textbf{Transition}\quad #2};
\node[tree node,minimum width=.92\linewidth,text width=.86\linewidth,below=of t] (b)
  {\textbf{Base Case}\quad #3};
\node[tree node,minimum width=.92\linewidth,text width=.86\linewidth,below=of b] (o)
  {\textbf{Evaluation Order}\quad #4};
\node[tree node,minimum width=.92\linewidth,text width=.86\linewidth,below=of o] (a)
  {\textbf{Answer}\quad #5};
\draw[dp edge] (s)--(t);\draw[dp edge] (t)--(b);
\draw[dp edge] (b)--(o);\draw[dp edge] (o)--(a);
\end{tikzpicture}}
