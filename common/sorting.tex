\newcommand{\heaparraytree}[3]{%
  \foreach \i in {1,...,10}{
    \pgfmathtruncatemacro{\lv}{floor(ln(\i)/ln(2))}
    \pgfmathtruncatemacro{\fst}{2^\lv}
    \pgfmathsetmacro{\gap}{8/(2^\lv)}
    \pgfmathsetmacro{\xx}{\gap/2+(\i-\fst)*\gap}
    \pgfmathsetmacro{\yy}{.65-.88*\lv}
    \coordinate (hc\i) at (\xx,\yy);
  }
  \foreach \i in {2,...,10}{\pgfmathtruncatemacro{\pp}{floor(\i/2)}\draw[AlgoRule,thick] (hc\pp)--(hc\i);}
  \foreach \v [count=\i] in {#1}{
    \ifnum\i>#3\relax
      \node[heap done] (h\i) at (hc\i) {\v};
      \node[sort done,minimum width=7.5mm] at ({.82*(\i-1)},2.15) {\v};
    \else
      \ifnum\i=#2\relax
        \node[heap current] (h\i) at (hc\i) {\v};
        \node[sort current,minimum width=7.5mm] at ({.82*(\i-1)},2.15) {\v};
      \else
        \node[heap node] (h\i) at (hc\i) {\v};
        \node[sort active,minimum width=7.5mm] at ({.82*(\i-1)},2.15) {\v};
      \fi
    \fi
    \node[font=\small,text=AlgoGray] at ({.82*(\i-1)},1.66) {\i};
  }
}
