\subsection{Naive Matching}
\begin{frame}{Naive Matching Pseudocode}
\algofont\begin{algorithmic}[1]
\Procedure{NaiveMatch}{$T,P$}\State \(n\gets|T|,\ m\gets|P|,\ result\gets[]\)
\If{\(m=0\)}\State\Return policy-defined result\EndIf
\For{\(s\gets0\) to \(n-m\)}\State \(j\gets0\)
\While{\(j<m\) \textbf{ and } \(T[s+j]=P[j]\)}\State \(j\gets j+1\)\EndWhile
\If{\(j=m\)}\State append \(s\) to \(result\)\EndIf\EndFor
\State\Return \(result\)\EndProcedure
\end{algorithmic}
\end{frame}

\begin{block}{표식}\(\checkmark\): match,\quad \(\times\): mismatch,\quad border: current comparison.\end{block}
\end{frame}
\begin{frame}{Alignment 수와 Comparison 수}
\[
\#alignments=n-m+1,\qquad
\#comparisons\le(n-m+1)m.
\]
\begin{itemize}\item alignment 하나가 곧 comparison 하나는 아니다.\item mismatch 위치에 따라 같은 alignment의 비용이 달라진다.\end{itemize}
\end{frame}

\begin{frame}{Naive Complexity}
\[
\text{worst } \Theta((n-m+1)m)=O(nm).
\]
\begin{itemize}
\item best: 각 alignment 첫 문자에서 mismatch
\[
\Theta(n-m+1)=O(n).
\]
\item \(m\)이 고정이거나 \(m\le cn\)인 상수 \(c<1\)이면 \(\Theta(n)\)으로 쓸 수 있다.
\item space: \(\Theta(1)\), result 제외
\end{itemize}
\end{frame}
