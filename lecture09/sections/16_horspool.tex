\subsection{Boyer--Moore--Horspool}
\begin{frame}{BuildShiftTable}
\scriptsize\begin{algorithmic}[1]
\Procedure{BuildShiftTable}{$P$}\State \(m\gets|P|\)
\ForAll{characters \(c\)}\State \(shift[c]\gets m\)\EndFor
\For{\(j\gets0\) to \(m-2\)}\State \(shift[P[j]]\gets m-1-j\)\EndFor
\State\Return \(shift\)\EndProcedure
\end{algorithmic}
\begin{block}{Overwrite}같은 문자가 반복되면 더 오른쪽 occurrence가 나중에 더 작은 distance로 덮어쓴다.\end{block}
\end{frame}

\begin{frame}{Horspool Search Pseudocode}
\fontsize{7}{7.6}\selectfont
\begin{algorithmic}[1]
\Procedure{Horspool}{$T,P$}
\State \(n\gets|T|;\ m\gets|P|;\ result\gets[]\)
\If{\(m=0\)}\State\Return policy-defined result\EndIf
\If{\(m>n\)}\State\Return \(result\)\EndIf
\State \(shift\gets\Call{BuildShiftTable}{P};\ s\gets0\)
\While{\(s\le n-m\)}
  \State \(j\gets m-1\)
  \While{\(j\ge0\) \textbf{ and } \(P[j]=T[s+j]\)}
    \State \(j\gets j-1\)
  \EndWhile
  \If{\(j<0\)}
    \State append \(s\) to \(result\)
  \EndIf
  \Statex \(\triangleright\) \textit{every alignment advances, match or mismatch}
  \State \(s\gets s+shift[T[s+m-1]]\)
\EndWhile
\State\Return \(result\)
\EndProcedure
\end{algorithmic}
\normalsize
\vspace{-1mm}
\begin{block}{Progress invariant}\small \(shift[c]\ge1\)이므로 mismatch도 \(s\)를 반드시 증가시키며 infinite loop가 불가능하다.\end{block}
\end{frame}

\begin{frame}{Match 후 Shift와 Overlap}
\[
T=\texttt{AAAAA},\quad P=\texttt{AAA},\quad shift[A]=1.
\]
match 후에도 현재 window 마지막 character \(A\)를 조회하므로 shifts \(0,1,2\)를 모두 방문하고 overlap을 놓치지 않는다.
\end{frame}
\begin{frame}{Horspool Complexity}
\small\begin{itemize}
\item preprocessing: dense alphabet \(\Theta(|\Sigma|+m)\), map은 expected \(\Theta(m)\)
\item search worst: \(\Theta(nm)\)
\item 큰 alphabet·긴 pattern·오른쪽 mismatch가 잦으면 실제 검사 위치를 많이 건너뜀
\end{itemize}
\begin{alertblock}{정확한 표현}검사한 text position 수가 \(n\)보다 작을 수 있다는 뜻이지, running time이 \(O(n)\)보다 작은 점근 bound라는 뜻은 아니다.\end{alertblock}
\end{frame}
\begin{frame}{Horspool Takeaway}
\stakeaway{Horspool은 window 마지막 character의 shift table과 right-to-left comparison을 결합한 실용적 변형이다.}
\begin{center}
\small pattern \(\rightarrow\) shift table \(\rightarrow\) mismatch
\(\rightarrow\) positive shift \(\rightarrow\) next alignment
\end{center}
\scheckpoint{\texttt{TIGER}에서 absent character는 5칸, \(I\)는 3칸 shift한다.}
\end{frame}
