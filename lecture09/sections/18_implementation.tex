\subsection{구현·테스트}
\begin{frame}{Java API}
\small\begin{itemize}
\item \texttt{naiveAll}, \texttt{rabinKarpAll}, \texttt{buildLps}, \texttt{kmpAll}
\item \texttt{buildHorspoolShift}, \texttt{horspoolAll}
\item first-match wrappers return \(-1\)
\item empty pattern all-match policy: \([0]\)
\item main comparison loops allocate no substring
\end{itemize}
\end{frame}
\begin{frame}{C API}
\small\begin{itemize}
\item explicit \texttt{unsigned char *} and \texttt{size\_t} lengths
\item byte-oriented exact semantics
\item \texttt{SIZE\_MAX} first-match sentinel
\item growable result with allocation-failure propagation
\item portable overflow-safe modular add/subtract/multiply
\end{itemize}
\end{frame}
