\begin{frame}{Text · Pattern · Alphabet}
\[
T[0..n-1],\qquad P[0..m-1],\qquad T[i],P[j]\in\Sigma.
\]
\begin{description}
\item[Text index \(i\)]현재 읽는 text 위치
\item[Pattern index \(j\)]현재 비교하는 pattern 위치
\item[Shift \(s\)]pattern 시작을 \(T[s]\)에 맞춘 alignment
\end{description}
\stakeaway{구현과 trace는 0-based indexing으로 통일한다.}
\end{frame}
\begin{frame}{Exact Occurrence의 정의}
\[
P\text{ occurs at }s
\iff
\forall j\in\{0,\ldots,m-1\},\quad T[s+j]=P[j].
\]
\[
0\le s\le n-m,\qquad\text{alignment 수}=n-m+1\quad(m\le n).
\]
\begin{alertblock}{구분}alignment는 검사할 후보 위치이고, occurrence는 모든 문자가 실제로 일치한 alignment다.\end{alertblock}
\end{frame}
