\begin{frame}{Ancestor · Descendant · Subtree}
\begin{columns}[T]\column{.48\textwidth}
\begin{block}{Transitive relation}A는 E의 ancestor, E는 A의 descendant다. parent/child를 여러 edge로 확장한 관계다.\end{block}
\column{.48\textwidth}
\begin{block}{Subtree}node B와 모든 descendant $\{B,D,E\}$, 그리고 그 edge로 이루어진 tree다.\end{block}
\end{columns}
\sttakeaway{Node degree는 child 수이고, tree degree는 모든 node degree의 최댓값이다.}
\end{frame}
\begin{frame}{Depth · Level · Height Convention}
\[
\operatorname{depth}(v)=\text{root에서 }v\text{까지 edge 수}
\]
\begin{itemize}
\item root depth $=0$, level $=\text{depth}+1$
\item node height $=$ 가장 깊은 descendant leaf까지 edge 수
\item leaf height $=0$, empty tree height $=-1$
\item tree height $=$ root height
\end{itemize}
\begin{alertblock}{Convention 확인}교재에 따라 level과 height를 node 수로 세기도 한다. 이 강의는 edge-height를 끝까지 사용한다.\end{alertblock}
\end{frame}
\begin{frame}{Tree의 세 가지 성질}
node가 $n\ge1$개인 tree는 다음을 만족한다.
\begin{enumerate}
\item connected이며 cycle이 없다.
\item edge가 정확히 $n-1$개다.
\item 임의의 두 node 사이에 unique simple path가 있다.
\end{enumerate}
\sttakeaway{세 문장은 서로 연결된다. edge를 하나 추가하면 cycle, 하나 제거하면 disconnect가 생긴다.}
\end{frame}
