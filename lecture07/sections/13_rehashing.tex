\subsection{Resize, Rehashing, Amortized Analysis}
\begin{frame}{Resize와 Rehash}
\begin{description}
\item[resize] table capacity를 바꾼다.
\item[rehash] active entries를 새 capacity 기준으로 다시 배치한다.
\end{description}
\begin{enumerate}
\item capacity가 큰 새 table 할당
\item 새 compression으로 bucket index 재계산
\item 모든 active entry 재삽입하고 tombstone 제거
\item 검증 후 old table 폐기
\end{enumerate}
\httakeaway{logical key--value contents와 size는 보존하지만 physical bucket은 바뀐다. index가 capacity에 의존하므로 단순 memory copy가 아니다.}
\end{frame}

\begin{frame}{Capacity Growth Policy}
\begin{itemize}\item power-of-two doubling + bit mixing\item next prime capacity + modulo reduction\item implementation-specific growth factor\end{itemize}
\begin{alertblock}{공동 설계}capacity policy, hash mixing, index reduction, collision scheme을 독립적으로 정하면 안 된다.\end{alertblock}

\begin{frame}{Amortized Rehashing}
\begin{itemize}\item 각 individual resizing PUT은 현재 entry 수에 비례해 $\Theta(n)$이다.\item $n$번 PUT 동안 모든 migration의 합은 $O(n)$이다.\item migration의 PUT당 amortized additional cost는 $O(1)$이다.\end{itemize}
\begin{alertblock}{정확한 결론}Suitable hashing과 controlled load 아래 resize 포함 PUT은 amortized expected $\Theta(1)$이다. 모든 PUT이 worst-case $\Theta(1)$이라는 뜻은 아니다.\end{alertblock}
\end{frame}
\begin{frame}{Resize 후 반드시 보존할 Invariant}
\begin{enumerate}\item 모든 active entry lookup 성공\item size 보존\item 새 capacity와 index 일치\item duplicate 없음\item old tombstone 제거\item load factor 재계산\end{enumerate}
\end{frame}
\begin{frame}{Rehashing Checkpoint}
\begin{enumerate}\item resize가 copy만으로 끝나지 않는 이유는?\item $1+2+\cdots+n$의 geometric sum bound는?\item 한 번의 resize 비용과 amortized insertion 비용은?\end{enumerate}
\begin{exampleblock}{답}index가 capacity 의존; $<2n$; resizing PUT $\Theta(n)$, 전체 sequence에서는 PUT당 추가 amortized $O(1)$이고 기본 hashing까지 포함하면 amortized expected $\Theta(1)$.\end{exampleblock}
\end{frame}
