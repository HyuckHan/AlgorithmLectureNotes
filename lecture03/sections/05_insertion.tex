\begin{frame}{Insertion Sort의 Mission과 Invariant}
\mission{$A[1..i-1]$가 정렬되었다고 믿고 $A[i]$를 올바른 위치에 삽입한다.}
\begin{block}{Invariant}각 반복 시작 시 prefix $A[1..i-1]$는 원래 prefix의 정렬된 permutation이다.\end{block}
\end{frame}
\begin{frame}{Insertion Sort 의사코드}
\begin{algorithmic}[1]\Procedure{InsertionSort}{$A[1..n]$}\For{$i\gets2$ to $n$}\State $key\gets A[i]$; $j\gets i-1$\While{$j\ge1$ and $A[j]>key$}\State $A[j+1]\gets A[j]$; $j\gets j-1$\EndWhile\State $A[j+1]\gets key$\EndFor\EndProcedure\end{algorithmic}
\end{frame}

\begin{frame}{Inversion과 Adaptive 성능}
\[\text{inversion}=\#\{(i,j):i<j\ \land\ A[i]>A[j]\}\]
표준 구현에서 \textbf{element shift 수는 inversion 수 $I$와 정확히 같다}.
\[
T(n,I)=\Theta(n+I).
\]
\begin{itemize}\item best: $I=0\Rightarrow\Theta(n)$\item worst: $I=\Theta(n^2)\Rightarrow\Theta(n^2)$\item 조건 $A[j]>key$가 strict하므로 같은 key는 서로 지나치지 않아 stable이다.\end{itemize}
\end{frame}
