\begin{frame}{Quick Sort: Partition이 핵심}
\mission{pivot을 골라 $\le pivot$과 $>pivot$ 구간으로 분할하고 각 구간만 재귀 정렬한다.}
\[\boxed{\le p}\quad\boxed{p}\quad\boxed{>p}\]
합치기 단계에는 별도 merge가 필요 없다.
\end{frame}
\begin{frame}{QUICKSORT 의사코드}
\begin{algorithmic}[1]\Procedure{QuickSort}{$A,low,high$}\If{$low<high$}\State $p\gets\Call{Partition}{A,low,high}$\State \Call{QuickSort}{$A,low,p-1$}\State \Call{QuickSort}{$A,p+1,high$}\EndIf\EndProcedure\end{algorithmic}
\end{frame}
\begin{frame}{Lomuto PARTITION 의사코드}
\small\begin{algorithmic}[1]\Procedure{Partition}{$A,low,high$}\State $pivot\gets A[high]$; $i\gets low-1$\For{$j\gets low$ to $high-1$}\If{$A[j]\le pivot$}\State $i\gets i+1$; \Call{Swap}{$A[i],A[j]$}\EndIf\EndFor\State \Call{Swap}{$A[i+1],A[high]$}\State \Return $i+1$\EndProcedure\end{algorithmic}
\vfill\muted{Scan invariant: $A[low..i]\le pivot$,\;
$A[i+1..j-1]>pivot$,\; $A[j..high-1]$은 미처리,\; $A[high]$는 pivot.}
\end{frame}

\begin{frame}{Pivot 선택과 중복 Key}
\optional[선택 심화]\hfill
\begin{itemize}\item random pivot: adversarial pattern을 확률적으로 피한다.\item median-of-three: 작은 비용으로 극단 pivot을 줄인다.\item 현재 Lomuto의 $A[j]\le pivot$은 같은 key가 많을 때 한쪽으로 몰릴 수 있다.\item 이 경우 3-way partition $(<,=,>)$이 효과적이다.\end{itemize}
\end{frame}
\begin{frame}{Quick Sort Takeaway}
\begin{itemize}\item cache locality가 좋고 expected time이 $\Theta(n\log n)$이다.\item 기본 구현은 unstable; recursion stack은 expected $\Theta(\log n)$, worst $\Theta(n)$.\item worst-case 보장이 필요하면 Heap/Merge 또는 introspective sort를 고려한다.\end{itemize}
\sortingtakeaway{pivot의 최종 위치만 확정되며, 두 재귀 호출은 서로 겹치지 않는다.}
\muted{Merge와 Quick에서 반복해서 등장한 $n\log n$이 comparison sorting에서 어떤 의미를 갖는지 결정 트리로 확인한다.}
\end{frame}
