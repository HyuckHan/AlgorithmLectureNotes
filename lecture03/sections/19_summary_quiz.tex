\begin{frame}{선택 Checklist 요약}
\begin{enumerate}\item 먼저 표준 라이브러리를 고려한다.\item worst-case, stability, memory 요구를 확인한다.\item 입력 크기·분포·정렬 정도·key range를 확인한다.\item comparator의 total order와 overflow 안전성을 검증한다.\end{enumerate}
\end{frame}
\begin{frame}{자주 하는 실수}
\begin{itemize}\item Big-O 하나만 보고 실제 전제와 worst/expected를 생략\item stable을 “값이 같은 원소를 구별하지 않음”으로 오해\item Heap array/tree index를 0-based와 1-based로 혼용\item BUILD-MAX-HEAP을 $\Theta(n\log n)$이라고 단정\item Quick Sort에서 pivot 양쪽이 아닌 pivot까지 다시 재귀 호출\item comparator에서 단순 뺄셈 사용\end{itemize}
\end{frame}

\begin{frame}{Quiz 정답 I: Trace}
\begin{itemize}
\item Selection: $[2_b,2_a,4,5]$로 equal records가 역전될 수 있다. Bubble: $[2_a,4,2_b,5]$. Insertion: 5와 4만 shift하고 $2_a$는 움직이지 않는다.
\item Merge의 $A[low],A[low+1],A[low+2]$는 $2,3,5$.
\item Lomuto의 $p$는 pivot의 최종 index이며 왼쪽은 $\le pivot$, 오른쪽은 $>pivot$이다.
\item Heap index 4의 parent는 2, children은 8·9다.
\item HEAPIFY 결과는 $[14,8,10,4,7,9,3,2,1,0]$이다.
\end{itemize}
\end{frame}
\begin{frame}{Quiz 정답 II · 다음 강의}
\begin{itemize}
\item 비교 하한 $\Omega(n\log n)$; BUILD-MAX-HEAP $\Theta(n)$
\item Stable: Merge·Insertion; $k=\Theta(n)$인 Counting: $\Theta(n)$
\item Counting은 작은 finite integer range, LSD Radix는 digit 표현과 stable pass가 필요하다.
\item C: relational comparison \texttt{(x>y)-(x<y)}
\item Java: \texttt{.thenComparing(f -> f.name)}
\end{itemize}
