\begin{frame}{Matrix Path DP 설계 Template}
\small
\dpdesign
{$dp[i][j]$: $(1,1)$에서 $(i,j)$까지의 최소 경로 합}
{$M[i][j]+\min(dp[i-1][j],dp[i][j-1])$}
{시작 cell, 첫 row 누적합, 첫 column 누적합}
{위·왼쪽 dependency 뒤: row-major 또는 column-major}
{$dp[n][m]$; 경로는 predecessor를 역추적}
\end{frame}
\begin{frame}{State와 Dependency}
\dpstate{$dp[i][j]$: $(1,1)$에서 $(i,j)$까지 이동하는 최소 경로 합}

dp[i][j]=M[i][j]+\min\{dp[i-1][j],dp[i][j-1]\}
\]

\begin{frame}{Base Cases가 경계를 결정한다}
\[
\begin{aligned}
dp[1][1]&=M[1][1],\\
dp[1][j]&=dp[1][j-1]+M[1][j],\\
dp[i][1]&=dp[i-1][1]+M[i][1].
\end{aligned}
\]
\begin{alertblock}{잘못된 초기화}\texttt{if (i==1 || j==1) return M[i][j]}는 누적합이 아니라 현재 값만 반환한다.\end{alertblock}
\end{frame}
\begin{frame}{Naive Recursive MinPath}
\algofont\begin{algorithmic}[1]
\Procedure{MinPath}{$i,j$}
\If{$i=1$ \textbf{and} $j=1$}\State\Return $M[1][1]$\EndIf
\If{$i=1$}\State\Return \Call{MinPath}{$1,j-1$}+$M[1][j]$\EndIf
\If{$j=1$}\State\Return \Call{MinPath}{$i-1,1$}+$M[i][1]$\EndIf
\State\Return $\min(\Call{MinPath}{i-1,j},\Call{MinPath}{i,j-1})+M[i][j]$
\EndProcedure
\end{algorithmic}
\end{frame}

\begin{frame}{Memoized Matrix Path}
\small\begin{algorithmic}[1]
\Procedure{MinPathMemo}{$i,j$}
\If{$seen[i][j]$}\State\Return $memo[i][j]$\EndIf
\State 정확한 boundary recurrence로 $memo[i][j]$ 계산
\State $seen[i][j]\gets true$; \Return $memo[i][j]$
\EndProcedure
\end{algorithmic}
\[
\underbrace{nm}_{\text{reachable states}}\times
\underbrace{\Theta(1)}_{\text{two predecessors}}=\Theta(nm)
\]
\begin{itemize}\item memo table $\Theta(nm)$, recursion stack $\Theta(n+m)$\item 음수 값 가능 시 `-1` sentinel 대신 visited 사용\end{itemize}
\end{frame}
\begin{frame}{Bottom-Up Evaluation Order}
\small\begin{algorithmic}[1]
\State $dp[1][1]\gets M[1][1]$
\For{$j\gets2$ \textbf{to} $m$}\State $dp[1][j]\gets dp[1][j-1]+M[1][j]$\EndFor
\For{$i\gets2$ \textbf{to} $n$}\State $dp[i][1]\gets dp[i-1][1]+M[i][1]$\EndFor
\For{$i\gets2$ \textbf{to} $n$}\For{$j\gets2$ \textbf{to} $m$}
\State $dp[i][j]\gets M[i][j]+\min(dp[i-1][j],dp[i][j-1])$
\EndFor\EndFor
\end{algorithmic}
\dptakeaway{row-major 순서에서는 위쪽 row와 현재 row의 왼쪽 cell이 모두 먼저 계산된다.}
\end{frame}

\begin{frame}{1-Row Space Optimization}
\[
dp[j]\gets M[i][j]+\min(\underbrace{dp[j]}_{\text{update 전: 위}},
\underbrace{dp[j-1]}_{\text{update 후: 왼쪽}})
\]
\begin{itemize}\item left-to-right update가 필수\item time $\Theta(nm)$, auxiliary space $\Theta(m)$\item path 복원이 필요하면 parent 또는 재계산 전략이 필요\end{itemize}
\end{frame}
